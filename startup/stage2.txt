# Стартовый скрипт этапа 2: команды-заглушки и обработка ошибок.
# Пустые строки пропускаются, строки с ошибками не прерывают скрипт.

ls
ls -l /home/user
cd /tmp
cd

# Ошибочные строки: выполнение продолжается
unknown-command arg1 arg2
exit not-a-number
exit 1 2
cd ..  # комментарий в конце строки
