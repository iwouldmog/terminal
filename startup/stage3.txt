# Стартовый скрипт этапа 3: все команды этапов 1-3 и работа с VFS.
# Внимание: vfs-init очищает физический ZIP-архив VFS.

# Команды ls и cd

no-such-command
exit abc
exit 1 2
vfs-init extra-argument

# Замена VFS на VFS по умолчанию (пустую)
vfs-init
ls
